definition{
  name: requestAllowed,
  parameters: [state : State, request : Request, grant : Grant, now : Nat],
  result: Bool,
  body: ((browserSupported (project request capability)) && ((project grant authorized) && (((project request session) =? (project state session)) && (((project grant session) =? (project state session)) && (((capabilityCode (project request capability)) =? (capabilityCode (project grant capability))) && ((Stitch.Identity::digestWellFormed (project request destinationRoot)) && ((Stitch.Identity::digestWellFormed (project request subjectRoot)) && ((Stitch.Identity::sameDigest (project request destinationRoot) (project grant destinationRoot)) && ((Stitch.Identity::sameDigest (project request subjectRoot) (project grant subjectRoot)) && (((project grant validFrom) <=? now) && ((now <? (project grant validUntil)) && ((if (requiresDisclosure (project request capability)) then (project grant disclosureAccepted) else true) && (Stitch.Budget::reserveLedgerAllowed (project state ledger) (project request actionId) (project request cost) now)))))))))))))

definition{
  name: reserve,
  parameters: [state : State, request : Request, grant : Grant, now : Nat],
  result: State,
  body: (record State {ledger := (if (requestAllowed state request grant now) then (Stitch.Budget::reserveLedger (project state ledger) (project request actionId) (project request cost) now) else (project state ledger)), session := (project state session), effects := (if (requestAllowed state request grant now) then (cons (record Effect {request := request, requestedAt := now, outcome := (ctor Outcome.Pending)}) (project state effects)) else (project state effects)), observations := (project state observations)})
}

definition{
  name: observationMatches,
  parameters: [effect : Effect, observation : Observation],
  result: Bool,
  body: (((outcomeCode (project effect outcome)) =? 0) && ((not ((outcomeCode (project observation outcome)) =? 0)) && ((Stitch.Identity::sameDigest (project (project effect request) actionId) (project observation actionId)) && ((Stitch.Identity::sameDigest (project (project effect request) destinationRoot) (project observation destinationRoot)) && ((Stitch.Identity::sameDigest (project (project effect request) subjectRoot) (project observation subjectRoot)) && (((capabilityCode (project (project effect request) capability)) =? (capabilityCode (project observation capability))) && (((project (project effect request) session) =? (project observation session)) && ((project effect requestedAt) <=? (project observation observedAt)))))))))
}

definition{
  name: pendingMatches,
  parameters: [effects : (List Effect), observation : Observation],
  result: Bool,
  body: (match effects
    | List.nil => false
    | List.cons head tail => ((observationMatches head observation) || (pendingMatches tail observation))),
  recursive_argument: effects
}

definition{
  name: finish,
  parameters: [effects : (List Effect), observation : Observation],
  result: (List Effect),
  body: (match effects
    | List.nil => (nil Effect)
    | List.cons head tail => (if (observationMatches head observation) then (cons (record Effect {request := (project head request), requestedAt := (project head requestedAt), outcome := (project observation outcome)}) tail) else (cons head (finish tail observation)))),
  recursive_argument: effects
}

definition{
  name: sessionFilter,
  parameters: [difference : Nat, whenEqual : State, whenDifferent : State],
  result: State,
  body: (match difference | Nat.zero => whenEqual | Nat.succ rest => whenDifferent)

theorem{
  name: sessionFilterLedger,
  parameters: [difference : Nat, whenEqual : State, whenDifferent : State],
  statement: ((project (sessionFilter difference whenEqual whenDifferent) ledger) = (ledgerFilter difference (project whenEqual ledger) (project whenDifferent ledger))),

theorem{
  name: positiveSessionFilterFact,
  parameters: [difference : Nat, whenEqual : State, whenDifferent : State],
  statement: (((0 < difference) -> ((sessionFilter difference whenEqual whenDifferent) = whenDifferent)) <-> (0 = 0)),
  proof: cases{
    branches: [
      {
        proof: constructor{
          branches: [linear_arithmetic{}, linear_arithmetic{definitions: [sessionFilter]}]

        binders: [],
        constructor: zero
      },

        proof: constructor{
          branches: [linear_arithmetic{}, linear_arithmetic{definitions: [sessionFilter]}]
        },
        binders: [rest],
        constructor: succ
      }
    ],
    scrutinee: difference
  },
  axioms: [Classical.choice, Quot.sound, propext]
}

definition{

  result: State,
  body: (record State {ledger := (project state ledger), session := (project state session), effects := (if (pendingMatches (project state effects) observation) then (finish (project state effects) observation) else (project state effects)), observations := (if (pendingMatches (project state effects) observation) then (cons observation (project state observations)) else (project state observations))})
