definition{
  name: acknowledgeTransfer,
  parameters: [transfer : Transfer, receipt : Receipt, now : Nat],
  result: Transfer,
  body: (match (project transfer phase)
    | TransferPhase.Requested => (if (receiptBound transfer receipt now) then (record Transfer {requestRoot := (project transfer requestRoot), manifestRoot := (project transfer manifestRoot), fragmentRoot := (project transfer fragmentRoot), sourceHolder := (project transfer sourceHolder), targetHolder := (project transfer targetHolder), index := (project transfer index), placementRevision := (project transfer placementRevision), phase := (ctor TransferPhase.Acknowledged), receipt := receipt}) else transfer)
    | TransferPhase.Acknowledged => transfer
    | TransferPhase.Committed => transfer
    | TransferPhase.Failed => transfer
    | TransferPhase.Uncertain => transfer)
}

definition{
  name: commitTransfer,
  parameters: [
    transfer : Transfer,
    now : Nat,

    targetDistinct : Bool

  result: Transfer,
  body: (match (project transfer phase)
    | TransferPhase.Requested => transfer

    | TransferPhase.Committed => transfer
    | TransferPhase.Failed => transfer
    | TransferPhase.Uncertain => transfer)
}
