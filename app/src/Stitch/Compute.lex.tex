definition{
  name: instructionAllowed,
  parameters: [opcode : Nat],
  result: Bool,
  body: ((opcode <=? 13) || ((opcode =? 15) || (opcode =? 16)))
}

definition{
  name: ringModulus,
  parameters: [bits : Nat],
  result: Nat,
  body: (match bits | Nat.zero => 1 | Nat.succ rest => (prim multiply 2 (ringModulus rest) : Nat)),
  recursive_argument: bits
}

definition{
  name: normalize,
  parameters: [value : Nat, modulus : Nat],
  result: Nat,
  body: (prim remainder value modulus 0 : Nat)
}

definition{
  name: natBitwise,
  parameters: [fuel : Nat, opcode : Nat, left : Nat, right : Nat],
  result: Nat,
  body: (match fuel
    | Nat.zero => 0
    | Nat.succ rest => (if ((left =? 0) && (right =? 0)) then 0 else (let lowLeft : Nat := (prim remainder left 2 0 : Nat); (let lowRight : Nat := (prim remainder right 2 0 : Nat); (let low : Nat := (if (opcode =? 7) then (if (lowLeft =? lowRight) then 0 else 1) else (if (opcode =? 8) then (prim multiply lowLeft lowRight : Nat) else (if ((lowLeft + lowRight) =? 0) then 0 else 1))); (low + (prim multiply 2 (natBitwise rest opcode (prim quotient left 2 0 : Nat) (prim quotient right 2 0 : Nat)) : Nat))))))),
  recursive_argument: fuel
}

definition{
  name: ringStep,
  parameters: [instruction : Instruction, value : Nat, modulus : Nat],
  result: Nat,
  body: (let x : Nat := (normalize value modulus); (let y : Nat := (normalize (project instruction argument) modulus); (let op : Nat := (project instruction opcode); (normalize (if (op =? 0) then (prim subtract modulus x : Nat) else (if (op =? 1) then (prim subtract (prim subtract modulus 1 : Nat) x : Nat) else (if (op =? 2) then (x + 1) else (if (op =? 3) then (prim subtract (x + modulus) 1 : Nat) else (if (op =? 4) then (x + y) else (if (op =? 5) then (prim subtract (x + modulus) y : Nat) else (if (op =? 6) then (prim multiply x y : Nat) else (if (op =? 7) then (natBitwise modulus 7 x y) else (if (op =? 8) then (natBitwise modulus 8 x y) else (if (op =? 9) then (natBitwise modulus 9 x y) else (if (op =? 10) then (if (x <=? y) then 1 else 0) else (if (op =? 11) then (if (x <? y) then 1 else 0) else (if (op =? 12) then (if (y <=? x) then 1 else 0) else (if (op =? 13) then (if (y <? x) then 1 else 0) else (if (op =? 15) then (prim quotient x y 0 : Nat) else (if (op =? 16) then (prim remainder x y 0 : Nat) else x)))))))))))))))) modulus))))
}

definition{
  name: execute,
  parameters: [instructions : (List Instruction), value : Nat, modulus : Nat],
  result: Nat,
  body: (match instructions
    | List.nil => value
    | List.cons head tail => (execute tail (ringStep head value modulus) modulus)),
  recursive_argument: instructions
}

definition{
  name: appendProgram,
  parameters: [prefix : (List Instruction), suffix : (List Instruction)],
  result: (List Instruction),
  body: (match prefix
    | List.nil => suffix
    | List.cons head tail => (cons head (appendProgram tail suffix))),
  recursive_argument: prefix
}

definition{
  name: checkpointLinkAllowed,
  parameters: [
    jobMatches : Bool,
    programMatches : Bool,
    inputMatches : Bool,
    modelMatches : Bool,
    predecessorMatches : Bool,
    replayMatches : Bool,
    currentStep : Nat,
    nextStep : Nat,
    programLength : Nat
  ],
  result: Bool,
  body: (replayMatches && (jobMatches && (programMatches && (inputMatches && (modelMatches && (predecessorMatches && ((currentStep <? nextStep) && (nextStep <=? programLength))))))))
}

theorem{
  name: semanticTamperingRefused,
  parameters: [
    jobMatches : Bool,
    programMatches : Bool,
    inputMatches : Bool,
    modelMatches : Bool,
    predecessorMatches : Bool,
    currentStep : Nat,
    nextStep : Nat,
    programLength : Nat
  ],
  statement: ((checkpointLinkAllowed jobMatches programMatches inputMatches modelMatches predecessorMatches false currentStep nextStep programLength) = false),
  proof: reflexivity{}
}

theorem{
  name: resumeEqualsUninterrupted,
  parameters: [
    prefix : (List Instruction),
    suffix : (List Instruction),
    value : Nat,
    modulus : Nat
  ],
  statement: ((execute (appendProgram prefix suffix) value modulus) = (execute suffix (execute prefix value modulus) modulus)),
  proof: induction{
    branches: [
      {proof: simplify{definitions: [appendProgram, execute]}, binders: [], constructor: nil},
      {
        proof: simplify{definitions: [appendProgram, execute, hypothesis]},
        binders: [head, tail, hypothesis],
        constructor: cons
      }
    ],
    generalizing: [value],
    scrutinee: prefix
  },
  axioms: [propext]
}

theorem{
  name: checkpointStrictlyAdvances,
  parameters: [
    jobMatches : Bool,
    programMatches : Bool,
    inputMatches : Bool,
    modelMatches : Bool,
    predecessorMatches : Bool,
    replayMatches : Bool,

    nextStep : Nat,

  statement: (((checkpointLinkAllowed jobMatches programMatches inputMatches modelMatches predecessorMatches replayMatches currentStep nextStep programLength) = true) -> (currentStep < nextStep)),
  proof: linear_arithmetic{definitions: [checkpointLinkAllowed]},
  axioms: [Quot.sound, propext]
}

definition{
  name: supportedWidth,
  parameters: [bits : Nat],
  result: Bool,
  body: ((bits =? 8) || ((bits =? 16) || ((bits =? 32) || ((bits =? 64) || (bits =? 128)))))
}

definition{
  name: instructionsAllowed,
  parameters: [instructions : (List Instruction)],
  result: Bool,
  body: (match instructions
    | List.nil => true
    | List.cons head tail => ((instructionAllowed (project head opcode)) && (instructionsAllowed tail))),
  recursive_argument: instructions
}

definition{
  name: instructionCount,
  parameters: [instructions : (List Instruction)],
  result: Nat,
  body: (match instructions | List.nil => 0 | List.cons head tail => (1 + (instructionCount tail))),
  recursive_argument: instructions
}

definition{
  name: sameInstruction,
  parameters: [left : Instruction, right : Instruction],
  result: Bool,
  body: (((project left opcode) =? (project right opcode)) && ((project left argument) =? (project right argument)))
}

definition{
  name: sameInstructions,
  parameters: [left : (List Instruction), right : (List Instruction)],
  result: Bool,
  body: (match left
    | List.nil => (match right | List.nil => true | List.cons head tail => false)
    | List.cons head tail => (match right
      | List.nil => false
      | List.cons other rest => ((sameInstruction head other) && (sameInstructions tail rest)))),
  recursive_argument: left
}

definition{
  name: instructionWellFormed,
  parameters: [instruction : Instruction, modulus : Nat],
  result: Bool,
  body: ((instructionAllowed (project instruction opcode)) && (if ((project instruction opcode) <=? 3) then ((project instruction argument) =? 0) else ((project instruction argument) <? modulus)))
}

definition{
  name: instructionsWellFormed,
  parameters: [instructions : (List Instruction), modulus : Nat],
  result: Bool,
  body: (match instructions
    | List.nil => true
    | List.cons head tail => ((instructionWellFormed head modulus) && (instructionsWellFormed tail modulus))),
  recursive_argument: instructions
}

definition{
  name: programAllowed,
  parameters: [program : Program, maximumSteps : Nat],
  result: Bool,
  body: ((supportedWidth (project program bits)) && ((Stitch.Identity::digestWellFormed (project program programRoot)) && ((Stitch.Identity::digestWellFormed (project program modelRoot)) && ((instructionsWellFormed (project program instructions) (ringModulus (project program bits))) && ((instructionCount (project program instructions)) <=? maximumSteps)))))
}

definition{
  name: takeSteps,
  parameters: [instructions : (List Instruction), count : Nat],
  result: (List Instruction),
  body: (match count
    | Nat.zero => (nil Instruction)
    | Nat.succ rest => (match instructions
      | List.nil => (nil Instruction)
      | List.cons head tail => (cons head (takeSteps tail rest)))),
  recursive_argument: count
}

definition{
  name: replayedValue,
  parameters: [job : Job, checkpoint : Checkpoint],
  result: Nat,
  body: (execute (takeSteps (project (project job program) instructions) (project checkpoint step)) (project job initialValue) (ringModulus (project (project job program) bits)))
}

definition{
  name: jobAllowed,
  parameters: [job : Job, maximumSteps : Nat],
  result: Bool,
  body: ((programAllowed (project job program) maximumSteps) && ((Stitch.Identity::digestWellFormed (project job jobRoot)) && ((Stitch.Identity::digestWellFormed (project job inputRoot)) && ((project job initialValue) <? (ringModulus (project (project job program) bits))))))
}

definition{
  name: checkpointMatchesJob,
  parameters: [job : Job, checkpoint : Checkpoint],
  result: Bool,
  body: ((jobAllowed job (instructionCount (project (project job program) instructions))) && ((Stitch.Identity::digestWellFormed (project checkpoint jobRoot)) && ((Stitch.Identity::digestWellFormed (project checkpoint programRoot)) && ((Stitch.Identity::digestWellFormed (project checkpoint inputRoot)) && ((Stitch.Identity::digestWellFormed (project checkpoint modelRoot)) && ((Stitch.Identity::digestWellFormed (project checkpoint priorRoot)) && ((Stitch.Identity::digestWellFormed (project checkpoint stateRoot)) && ((Stitch.Identity::digestWellFormed (project checkpoint checkpointRoot)) && ((Stitch.Identity::sameDigest (project job jobRoot) (project checkpoint jobRoot)) && ((Stitch.Identity::sameDigest (project (project job program) programRoot) (project checkpoint programRoot)) && ((Stitch.Identity::sameDigest (project job inputRoot) (project checkpoint inputRoot)) && ((Stitch.Identity::sameDigest (project (project job program) modelRoot) (project checkpoint modelRoot)) && (((project checkpoint step) <=? (instructionCount (project (project job program) instructions))) && ((project checkpoint value) =? (replayedValue job checkpoint)))))))))))))))
}

definition{
  name: checkpointSuccessor,
  parameters: [job : Job, current : Checkpoint, candidate : Checkpoint],
  result: Bool,
  body: ((checkpointMatchesJob job current) && ((checkpointMatchesJob job candidate) && ((Stitch.Identity::sameDigest (project candidate priorRoot) (project current checkpointRoot)) && ((project current step) <? (project candidate step)))))
}

theorem{

  parameters: [program : Program, observation : ProgramDigestObservation, maximumSteps : Nat],
  result: Bool,
  body: ((programAllowed program maximumSteps) && (((project (project observation input) version) =? 1) && ((Stitch.Identity::sameDigest (project (project observation input) modelRoot) (project program modelRoot)) && (((project (project observation input) bits) =?
