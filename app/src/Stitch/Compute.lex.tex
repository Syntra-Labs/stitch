definition{
  name: naturalPower,
  parameters: [fuel : Nat, base : Nat, exponent : Nat, accumulated : Nat],
  result: Nat,
  body: (match fuel
    | Nat.zero => accumulated
    | Nat.succ rest => (if (exponent =? 0) then accumulated else (naturalPower rest (prim multiply base base : Nat) (prim quotient exponent 2 0 : Nat) (if ((prim remainder exponent 2 0 : Nat) =? 1) then (prim multiply accumulated base : Nat) else accumulated)))),
  recursive_argument: fuel
}

definition{

  result: Nat,

    | Nat.zero => accumulated
    | Nat.succ rest => (if ((left =? 0) && (right =? 0)) then accumulated else (let lowLeft : Nat := (prim remainder left 2 0 : Nat); (let lowRight : Nat := (prim remainder right 2 0 : Nat); (let low : Nat := (if (opcode =? 7) then (if (lowLeft =? lowRight) then 0 else 1) else (if (opcode =? 8) then (prim multiply lowLeft lowRight : Nat) else (if ((lowLeft + lowRight) =? 0) then 0 else 1))); (bitwiseAcc rest opcode (prim quotient left 2 0 : Nat) (prim quotient right 2 0 : Nat) (prim multiply place 2 : Nat) (accumulated + (prim multiply low place : Nat)))))))),
  recursive_argument: fuel
}

definition{

  recursive_argument: fuel
}

    | Nat.zero => accumulated

}

definition{

definition{
  name: wordAllowed,

  result: Bool,

}

  parameters: [left : Nat, right : Nat],
  result: Nat,
  body: (if (left <=? right) then right else left)
}

definition{
  name: arithmeticWidth,
  parameters: [left : Nat, right : Nat],
  result: Nat,
  body: (operandWidth 8 (operandWidth left right))
}

definition{
  name: typedResultBits,
  parameters: [instruction : TypedInstruction, bits : Nat],
  result: Nat,
  body: (if ((project instruction opcode) =? 14) then (bits + (project instruction argumentBits)) else (if ((10 <=? (project instruction opcode)) && ((project instruction opcode) <=? 13)) then 8 else (arithmeticWidth bits (project instruction argumentBits))))
}

definition{
  name: typedInstructionAdmission,
  parameters: [instruction : TypedInstruction, bits : Nat, capacityBits : Nat, resultBits : Nat],
  result: Bool,
  body: ((resultBits <=? capacityBits) && ((byteWidth bits) && ((bits <=? capacityBits) && (if ((project instruction opcode) <=? 3) then (((project instruction argumentBits) =? 0) && ((project instruction argument) =? 0)) else ((byteWidth (project instruction argumentBits)) && (((project instruction argument) <? (ringModulus (project instruction argumentBits))) && (if ((project instruction opcode) =? 14) then ((bits + (project instruction argumentBits)) <=? capacityBits) else ((instructionAllowed (project instruction opcode)) && (((arithmeticWidth bits (project instruction argumentBits)) <=? capacityBits) && (if (((project instruction opcode) =? 15) || ((project instruction opcode) =? 16)) then (0 <? (project instruction argument)) else true))))))))))
}

definition{
  name: typedInstructionAllowed,
  parameters: [instruction : TypedInstruction, bits : Nat, capacityBits : Nat],

}

definition{
  name: typedProgramAllowed,
  parameters: [instructions : (List TypedInstruction), bits : Nat, capacityBits : Nat],
  result: Bool,
  body: (match instructions
    | List.nil => ((byteWidth bits) && (bits <=? capacityBits))
    | List.cons head tail => ((typedInstructionAllowed head bits capacityBits) && (typedProgramAllowed tail (typedResultBits head bits) capacityBits))),
  recursive_argument: instructions
}

definition{

  parameters: [left : Word, right : Word],

definition{
  name: typedStep,
  parameters: [instruction : TypedInstruction, word : Word],
  result: Word,
  body: (if ((project instruction opcode) =? 14) then (concatWords word (record Word {bits := (project instruction argumentBits), value := (project instruction argument)})) else (record Word {bits := (typedResultBits instruction (project word bits)), value := (ringStep (record Instruction {opcode := (project instruction opcode), argument := (project instruction argument)}) (project word value) (ringModulus (arithmeticWidth (project word bits) (project instruction argumentBits))))}))
}

definition{
  name: typedInstructionWork,
  parameters: [instruction : TypedInstruction, bits : Nat],
  result: Nat,
  body: (let width : Nat := (arithmeticWidth bits (project instruction argumentBits)); (let limbs : Nat := (prim quotient (width + 63) 64 0 : Nat); (let productWork : Nat := (prim multiply limbs limbs : Nat); (if ((project instruction opcode) =? 14) then (1 + (prim quotient (bits + (project instruction argumentBits)) 8 0 : Nat)) else (if ((project instruction opcode) =? 17) then (1 + (prim multiply productWork (powerWork width (project instruction argument)) : Nat)) else (if (((project instruction opcode) =? 6) || ((15 <=? (project instruction opcode)) && ((project instruction opcode) <=? 16))) then (1 + productWork) else (if ((7 <=? (project instruction opcode)) && ((project instruction opcode) <=? 9)) then (1 + width) else (1 + limbs))))))))
}

definition{
  name: typedProgramWorkAcc,
  parameters: [instructions : (List TypedInstruction), bits : Nat, accumulated : Nat],
  result: Nat,
  body: (match instructions
    | List.nil => accumulated
    | List.cons head tail => (typedProgramWorkAcc tail (typedResultBits head bits) (accumulated + (typedInstructionWork head bits)))),
  recursive_argument: instructions
}

definition{
  name: typedProgramWork,
  parameters: [instructions : (List TypedInstruction), bits : Nat],
  result: Nat,
  body: (typedProgramWorkAcc instructions bits 0)
}

definition{
  name: typedExecute,
  parameters: [instructions : (List TypedInstruction), word : Word],
  result: Word,
  body: (match instructions
    | List.nil => word
    | List.cons head tail => (typedExecute tail (typedStep head word))),
  recursive_argument: instructions
