definition{
  name: xorBit,
  parameters: [left : Nat, right : Nat],
  result: Nat,
  body: (if (left =? right) then 0 else 1)
}

definition{
  name: recoverLeftParity,
  parameters: [pair : BitPair],
  result: BitPair,
  body: (record BitPair {left := (project pair left), right := (xorBit (project pair left) (xorBit (project pair left) (project pair right)))})
}

definition{
  name: recoverRightParity,
  parameters: [pair : BitPair],
  result: BitPair,
  body: (record BitPair {left := (xorBit (xorBit (project pair left) (project pair right)) (project pair right)), right := (project pair right)})
}

theorem{
  name: leftParityBitExact,
  parameters: [left : Nat, right : Nat],
  statement: ((left <= 1) -> ((right <= 1) -> ((xorBit left (xorBit left right)) = right))),
  proof: linear_arithmetic{definitions: [xorBit]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: rightParityBitExact,
  parameters: [left : Nat, right : Nat],
  statement: ((left <= 1) -> ((right <= 1) -> ((xorBit (xorBit left right) right) = left))),
  proof: linear_arithmetic{definitions: [xorBit]},
  axioms: [Quot.sound, propext]
}

definition{
  name: pairAllowed,
  parameters: [
    leftIndex : Nat,
    rightIndex : Nat,
    manifestMatches : Bool,
    fragmentsValid : Bool,
    dependenciesPresent : Bool
  ],
  result: Bool,
  body: ((leftIndex <? 3) && ((rightIndex <? 3) && (((leftIndex <? rightIndex) || (rightIndex <? leftIndex)) && (manifestMatches && (fragmentsValid && dependenciesPresent)))))
}

definition{
  name: xorCell,
  parameters: [left : Bit, right : Bit],
  result: Bit,
  body: (match left
    | Bit.Zero => right
    | Bit.One => (match right | Bit.Zero => (ctor Bit.One) | Bit.One => (ctor Bit.Zero)))
}

definition{
  name: leftParity,
  parameters: [blocks : BitBlocks],
  result: BitBlocks,
  body: (match blocks
    | BitBlocks.End => (ctor BitBlocks.End)
    | BitBlocks.More left right tail => (ctor BitBlocks.More left (xorCell left right) (leftParity tail))),
  recursive_argument: blocks
}

definition{
  name: rightParity,
  parameters: [blocks : BitBlocks],
  result: BitBlocks,
  body: (match blocks
    | BitBlocks.End => (ctor BitBlocks.End)
    | BitBlocks.More left right tail => (ctor BitBlocks.More (xorCell left right) right (rightParity tail))),
  recursive_argument: blocks
}

inductive{
  name: PairSelection,
  type_parameters: [],
  parameters: [],
  constructors: [
    {name: Systematic, fields: []},
    {name: LeftParity, fields: []},
    {name: RightParity, fields: []}
  ]

definition{
  name: encodeForPair,
  parameters: [blocks : BitBlocks, selection : PairSelection],
  result: BitBlocks,
  body: (match selection
    | PairSelection.Systematic => blocks
    | PairSelection.LeftParity => (leftParity blocks)
    | PairSelection.RightParity => (rightParity blocks))
}

definition{
  name: decodePair,
  parameters: [available : BitBlocks, selection : PairSelection],
  result: BitBlocks,
  body: (match selection
    | PairSelection.Systematic => available
    | PairSelection.LeftParity => (leftParity available)
    | PairSelection.RightParity => (rightParity available))
}

theorem{
  name: everyCompatiblePairRecoversFiniteBlocks,
  parameters: [blocks : BitBlocks, selection : PairSelection],
  statement: ((decodePair (encodeForPair blocks selection) selection) = blocks),
  proof: cases{
    branches: [
      {
        proof: simplify{definitions: [decodePair, encodeForPair]},
        binders: [],
        constructor: Systematic

      {
        proof: simplify{definitions: [decodePair, encodeForPair, leftParityInvolutive]},
        binders: [],
        constructor: LeftParity

        binders: [],
        constructor: RightParity
      }

  },
