definition{
  name: sameIdentity,
  parameters: [left : Identity, right : Identity],
  result: Bool,
  body: ((prim equal (project left channel) (project right channel) : Bool) && ((prim equal (project left session) (project right session) : Bool) && ((prim equal (project left digest) (project right digest) : Bool) && ((prim equal (project left purpose) (project right purpose) : Bool) && (((project left created) =? (project right created)) && (((project left messageBytes) =? (project right messageBytes)) && ((project left frames) =? (project right frames))))))))
}

definition{
  name: frameAllowed,
  parameters: [
    bounds : Limits,
    frame : Frame,
    channel : Bytes,
    purpose : Bytes,
    wallNow : Nat,
    lifetime : Nat
  ],
  result: Bool,
  body: (let identity : Identity := (project frame identity); ((prim equal (project identity channel) channel : Bool) && ((prim equal (project identity purpose) purpose : Bool) && ((0 <? (project identity messageBytes)) && (((project identity messageBytes) <=? (project bounds messageBytes)) && ((0 <? (project identity frames)) && (((project identity frames) <=? (project bounds frames)) && (((project frame index) <? (project identity frames)) && (((project frame wireBytes) <=? (project bounds wireBytes)) && (((prim multiply (prim subtract (project identity frames) 1 : Nat) (project bounds chunkBytes) : Nat) <? (project identity messageBytes)) && (((project identity messageBytes) <=? (prim multiply (project identity frames) (project bounds chunkBytes) : Nat)) && (((project frame payloadBytes) =? (if (((project frame index) + 1) =? (project identity frames)) then (prim subtract (project identity messageBytes) (prim multiply (prim subtract (project identity frames) 1 : Nat) (project bounds chunkBytes) : Nat) : Nat) else (project bounds chunkBytes))) && (((project identity created) <=? (wallNow + (project bounds futureSkew))) && (wallNow <? ((project identity created) + lifetime)))))))))))))))
}

definition{
  name: admissionAllowed,
  parameters: [bounds : Limits, capacity : Capacity, bytes : Nat],
  result: Bool,
  body: ((0 <? bytes) && ((bytes <=? (project bounds messageBytes)) && (((project capacity entries) <? (project bounds entries)) && ((((project capacity retainedBytes) + bytes) <=? (project bounds retainedBytes)) && (((project capacity entries) + (project capacity tombstones)) <? (project bounds tombstones))))))
}

definition{
  name: admitCapacity,
  parameters: [bounds : Limits, capacity : Capacity, bytes : Nat],
  result: Capacity,
  body: (if (admissionAllowed bounds capacity bytes) then (record Capacity {entries := ((project capacity entries) + 1), retainedBytes := ((project capacity retainedBytes) + bytes), tombstones := (project capacity tombstones)}) else capacity)
}

definition{
  name: release,
  parameters: [capacity : Capacity, bytes : Nat],
  result: Capacity,
  body: (record Capacity {entries := (prim subtract (project capacity entries) 1 : Nat), retainedBytes := (prim subtract (project capacity retainedBytes) bytes : Nat), tombstones := ((project capacity tombstones) + 1)})
}

definition{
  name: assemblyCurrent,
  parameters: [bounds : Limits, state : Assembly, now : Nat],
  result: Bool,
  body: (((project state last) <=? now) && ((now <? ((project state started) + (project bounds assemblyAge))) && (not (project state verifying))))
}

definition{
  name: appendAllowed,
  parameters: [bounds : Limits, state : Assembly, frame : Frame, now : Nat, duplicate : Bool],
  result: Bool,
  body: ((not duplicate) && ((assemblyCurrent bounds state now) && ((sameIdentity (project state identity) (project frame identity)) && ((((project state receivedFrames) + 1) <=? (project (project state identity) frames)) && (((project state receivedBytes) + (project frame payloadBytes)) <=? (project (project state identity) messageBytes))))))
}

definition{
  name: append,
  parameters: [bounds : Limits, state : Assembly, frame : Frame, now : Nat, duplicate : Bool],
  result: Assembly,
  body: (if (appendAllowed bounds state frame now duplicate) then (record Assembly {identity := (project state identity), receivedFrames := ((project state receivedFrames) + 1), receivedBytes := ((project state receivedBytes) + (project frame payloadBytes)), started := (project state started), last := now, verifying := false}) else state)
}

definition{
  name: ready,
  parameters: [state : Assembly],
  result: Bool,
  body: (((project state receivedFrames) =? (project (project state identity) frames)) && ((project state receivedBytes) =? (project (project state identity) messageBytes)))
}

definition{
  name: beginVerification,
  parameters: [state : Assembly],
  result: Assembly,
  body: (record Assembly {identity := (project state identity), receivedFrames := (project state receivedFrames), receivedBytes := (project state receivedBytes), started := (project state started), last := (project state last), verifying := (ready state)})
}

definition{
  name: deliveryAllowed,
  parameters: [
    state : Assembly,
    digestVerified : Bool,
    utf8Valid : Bool,
    envelopeValid : Bool,
    sessionCurrent : Bool,
    consentCurrent : Bool
  ],
  result: Bool,
  body: (digestVerified && (utf8Valid && (envelopeValid && (sessionCurrent && (consentCurrent && ((project state verifying) && (ready state)))))))
}

definition{
  name: effectAllowed,
  parameters: [
    closed : Bool,
    currentGeneration : Nat,
    capturedGeneration : Nat,
    sessionMatches : Bool,
    consentCurrent : Bool
  ],
  result: Bool,
  body: ((not closed) && ((currentGeneration =? capturedGeneration) && (sessionMatches && consentCurrent)))
}

theorem{
  name: duplicateDoesNotAdvanceAssembly,
  parameters: [bounds : Limits, state : Assembly, frame : Frame, now : Nat],
  statement: ((append bounds state frame now true) = state),
  proof: reflexivity{}
}

theorem{
  name: unverifiedMessageCannotDeliver,
  parameters: [
    state : Assembly,
    utf8Valid : Bool,
    envelopeValid : Bool,
    sessionCurrent : Bool,
    consentCurrent : Bool
  ],

}

theorem{
  name: closedSessionCannotPublish,
  parameters: [
    currentGeneration : Nat,
    capturedGeneration : Nat,
    sessionMatches : Bool,
    consentCurrent : Bool
  ],

  proof: reflexivity{}
}

theorem{
  name: effectFalseGuard,
  parameters: [],
  statement: (forall (open : Bool), (forall (matching : Bool), (forall (session : Bool), ((open && (matching && (session && false))) = false)))),
  proof: decide{}
}

theorem{
  name: withdrawnConsentCannotPublish,

    closed : Bool,
    currentGeneration : Nat,

  ],

  proof: simplify{definitions: [effectAllowed, effectFalseGuard]},
  axioms: [propext]
}
