definition{
  name: dependenciesUnchanged,
  parameters: [
    left : (List Stitch.Encoding::Dependency),
    right : (List Stitch.Encoding::Dependency)
  ],
  result: Bool,
  body: (match left
    | List.nil => (match right | List.nil => true | List.cons other rest => false)
    | List.cons head tail => (match right
      | List.nil => false
      | List.cons other rest => ((sameDependency head other) && (dependenciesUnchanged tail rest)))),
  recursive_argument: left
}

definition{
  name: effectMember,
  parameters: [effect : HostEffect, effects : (List HostEffect)],
  result: Bool,
  body: (match effects
    | List.nil => false
    | List.cons head tail => (((effectCode effect) =? (effectCode head)) || (effectMember effect tail))),
  recursive_argument: effects
}

definition{
  name: effectsCovered,
  parameters: [requested : (List HostEffect), allowed : (List HostEffect)],
  result: Bool,
  body: (match requested
    | List.nil => true
    | List.cons head tail => ((effectMember head allowed) && (effectsCovered tail allowed))),
  recursive_argument: requested
}
