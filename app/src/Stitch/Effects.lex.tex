definition{

theorem{
  name: vmPositiveGenerationFilterFact,
  parameters: [difference : Nat, whenEqual : Vm, whenDifferent : Vm],
  statement: (((0 < difference) -> ((vmGenerationFilter difference whenEqual whenDifferent) = whenDifferent)) <-> (0 = 0)),
  proof: cases{
    branches: [
      {
        proof: constructor{
          branches: [linear_arithmetic{}, linear_arithmetic{definitions: [vmGenerationFilter]}]
        },
        binders: [],
        constructor: zero
      },
      {
        proof: constructor{
          branches: [linear_arithmetic{}, linear_arithmetic{definitions: [vmGenerationFilter]}]
        },
        binders: [rest],
        constructor: succ
      }
    ],
    scrutinee: difference
  },
  axioms: [Classical.choice, Quot.sound, propext]
}
