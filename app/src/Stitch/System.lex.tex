definition{
  name: reservationActive,
  parameters: [
    reservations : (List Stitch.Budget::Reservation),
    actionId : Stitch.Identity::Digest,
    epoch : Nat
  ],
  result: Bool,
  body: (match reservations
    | List.nil => false
    | List.cons head tail => (if (Stitch.Identity::sameDigest (project head actionId) actionId) then (((project head epoch) =? epoch) && (not (project head settled))) else (reservationActive tail actionId epoch))),
  recursive_argument: reservations
}
